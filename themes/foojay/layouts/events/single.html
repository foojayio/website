{{ define "main" }} {{/* /calendar/ -- the events calendar. Everything on this page is derived from data/events.json (written by scripts/FetchMeetupEvents.java, four times a day) joined to data/jugs.yaml on the group's `jug` slug for the country. There is nothing to author: the page file holds a one-line intro and this layout does the rest, the same way /jugs/ and /java-champions/ work. Two renderings of one set of events, and only one of them is content: the agenda list below is server-rendered (so it is what Pagefind indexes and what a reader without JavaScript gets), and the month grid is a visualisation of the same events built client-side from the JSON island. The toolbar starts `hidden` because every control on it is JS-only -- without the script the page is simply the agenda. */}} {{ $data := hugo.Data.events }} {{/* Join key: events.json's `jug` is a data/jugs.yaml slug -- verified, all 32 groups match -- which is where the country comes from. */}} {{ $jugBySlug := dict }} {{ range hugo.Data.jugs }} {{ $jugBySlug = merge $jugBySlug (dict .slug .) }} {{ end }} {{/* Flatten groups -> events, and enrich each one with everything the grid, the agenda and the detail dialog need, so neither the templates below nor the JavaScript has to re-derive it. Times are formatted here, in the event's OWN offset (that is what the JUG means by "19:00" and what Meetup shows); rebuilding them from the ISO string in the browser would silently restate them in the reader's zone instead. */}} {{ $events := slice }} {{ $errored := slice }} {{ range $g := $data.groups }} {{ if $g.error }} {{ $errored = $errored | append $g.name }} {{ end }} {{ $jug := index $jugBySlug $g.jug }} {{ $country := "" }} {{ with $jug }}{{ $country = .country | default "" }}{{ end }} {{/* One hue per group, derived from its slug, so a JUG keeps the same dot colour everywhere on the page without a colour ever being configured. */}} {{ $hue := mod (hash.FNV32a $g.jug) 360 }} {{ range $e := ($g.events | default slice) }} {{/* One string for where the event is: "Online", else venue + city as the JSON-LD gave them, else nothing (and the templates below fall back to the JUG's country). Built once here so the grid, the agenda and the dialog can't disagree about it. */}} {{ $place := "" }} {{ if eq $e.online true }} {{ $place = "Online" }} {{ else }} {{ $parts := slice }} {{ with $e.venue }}{{ $parts = $parts | append . }}{{ end }} {{ with $e.city }}{{ $parts = $parts | append . }}{{ end }} {{ $place = delimit $parts ", " }} {{ end }} {{ $start := time.AsTime $e.startTime }} {{ $endDate := "" }} {{ $endTime := "" }} {{ with $e.endTime }} {{ $end := time.AsTime . }} {{ $endDate = $end.Format "2006-01-02" }} {{ $endTime = $end.Format "15:04" }} {{ end }} {{ $events = $events | append (dict "title" $e.title "url" $e.url "group" $g.name "groupUrl" ($g.url | default "") "jug" $g.jug "country" $country "hue" $hue "sort" $start.Unix "date" ($start.Format "2006-01-02") "endDate" $endDate "day" ($start.Format "2") "weekday" ($start.Format "Mon") "time" ($start.Format "15:04") "endTime" $endTime "dateLabel" ($start.Format "Monday, 2 January 2006") "monthKey" ($start.Format "2006-01") "monthLabel" ($start.Format "January 2006") "venue" ($e.venue | default "") "city" ($e.city | default "") "online" (eq $e.online true) "place" $place ) }} {{ end }} {{ end }} {{ $events = sort $events "sort" }} {{/* Months present in the data, in order, plus their labels -- Hugo's `group` only works on pages, so the agenda's month headings are collected here and the rows filtered per month below. 25 events; the nested loop is free. */}} {{ $monthKeys := slice }} {{ $monthLabels := dict }} {{ $groupNames := slice }} {{ $countries := slice }} {{ range $events }} {{ if not (in $monthKeys .monthKey) }} {{ $monthKeys = $monthKeys | append .monthKey }} {{ $monthLabels = merge $monthLabels (dict .monthKey .monthLabel) }} {{ end }} {{ if not (in $groupNames .group) }}{{ $groupNames = $groupNames | append .group }}{{ end }} {{ if and .country (not (in $countries .country)) }}{{ $countries = $countries | append .country }}{{ end }} {{ end }}
