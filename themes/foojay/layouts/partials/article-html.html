{{- /* A page's rendered body, with asciidoctor's code blocks translated into the markup the rest of the site already uses. Takes the Page, returns HTML; a Markdown page gets .Content back untouched and costs nothing. Hugo's render hooks are a Goldmark feature, so _markup/render-codeblock.html never sees a .adoc article and rewriting asciidoctor's HTML afterwards is the only hook available. The language name goes through partials/code-language.html, the same registry the Markdown hook uses, so `[source,kt]` and ```kt cannot drift onto different lexers. CALLED FROM baseof.html TOO, and that is not optional: baseof decides whether to load mermaid.js by looking for `
`, which
       on a .adoc page does not exist until this has run. Both callers use
       partialCached on the same key, so the regex work happens once. */ -}}

{{- $page := . -}}
{{- $html := $page.Content -}}

{{- if and $page.File (eq $page.File.Ext "adoc") -}}

  {{- /* Diagrams first, for the same reason the Markdown hook takes them
         first: mermaid is not a language and must not reach the highlighter. */ -}}
  {{- range $block := findRE `(?s)
]*>.*?
` $html -}}
    {{- $inner := replaceRE `(?s)^]*>]*>(.*?)
$` "$1" $block -}} {{- $html = replace $html $block (printf `
%s
` $inner) -}} {{- end -}} {{- /* Then every other tagged block. */ -}} {{- range $block := findRE `(?s)
]*>.*?
` $html -}} {{- $tag := replaceRE `(?s)^]*>]*>]*>(.*?)$` "$1" $block -}} {{- $lang := partial "code-language.html" $tag -}} {{- $html = replace $html $block (printf `
%s
` $lang $inner) -}} {{- end -}} {{- /* And the untagged ones. A `----` block with no [source] attribute gets no wrapper and no class at all -- just a bare 
 inside the
         listingblock -- so it needs its own rule, anchored on that wrapper so
         the pattern cannot reach a 
 anywhere else on the page. `generic`
         is what an untagged ``` fence gets in Markdown, so the two agree. */ -}}
  {{- $html = replaceRE `(?s)
(.*?)
` `
$1
` $html -}}
  {{- $html = replaceRE `(?s)(
\s*
\s*)
(.*?)
` `${1}
$2
` $html -}}

  {{- /* LINKS. _markup/render-link.html is a Goldmark hook too, so an AsciiDoc
         article gets none of it and the same two rewrites have to happen here
         or the two formats disagree about what a link means. Only fires when
         the deploy has a baseURL SUBPATH -- a preview host -- since otherwise
         every link is already correct and this would rewrite 4000 pages' hrefs
         to identical values.

         RE2 has no negative lookahead, hence `([^/"])` to hold the character
         after the slash: it is what keeps a protocol-relative `//host/path`
         from being mangled into the site's own base path.

         relURL "" and NOT relURL "/": relURL leaves a path that already starts
         with a slash alone, so relURL "/" is "/" everywhere and the guard
         below would never fire. The link hook works around it the same way. */ -}}
  {{- $base := relURL "" -}}
  {{- if ne $base "/" -}}
    {{- $html = replaceRE `href="(?:https?:)?//(?:www\.)?foojay\.io/` (printf `href="%s` $base) $html -}}
    {{- $html = replaceRE `href="/([^/"])` (printf `href="%s$1` $base) $html -}}
  {{- end -}}

  {{- /* EXTERNAL LINKS OPEN IN A NEW TAB, as the link hook does for Markdown.

         AFTER the baseURL rewrite above, and that order is load-bearing: on a
         subpath deploy that rewrite has already turned an absolute foojay.io
         href root-relative, so it no longer matches here and correctly stays
         in the same tab. In production the rewrite does not fire, so such a
         link survives to this point and has to be excluded by host -- which is
         what $internal is for, and it is a COPY of the link hook's test so the
         two formats cannot disagree about which hosts are ours.

         TAG BY TAG rather than one replaceRE, because RE2 cannot express
         either skip without lookahead: a link that already carries a target
         (asciidoctor writes one for the `^` suffix, and rewriting it would
         emit a duplicate attribute), and our own host. */ -}}
  {{- range $tag := findRE `]*href="(?:https?:)?//[^"]*"[^>]*>` $html -}}
    {{- if not (strings.Contains $tag "target=") -}}
      {{- $href := replaceRE `^.*?href="([^"]*)".*$` "$1" $tag -}}
      {{- $internal := or (hasPrefix $href "https://foojay.io") (hasPrefix $href "http://foojay.io") (hasPrefix $href "https://www.foojay.io") (hasPrefix $href "http://www.foojay.io") (hasPrefix $href "//foojay.io") -}}
      {{- if not $internal -}}
        {{- $html = replace $html $tag (replaceRE `>$` ` target="_blank" rel="noopener noreferrer">` $tag) -}}
      {{- end -}}
    {{- end -}}
  {{- end -}}

  {{- /* CALLOUTS: `// <1>` at the end of a code line becomes a circled digit.
         Applied to the whole article, not per block, because an untagged
         `----` block carries callouts too.

         A CHARACTER is the only form that survives: EnlighterJS re-renders a
         block from its textContent, so asciidoctor's `` would
         reach the reader as escaped markup. U+2460 up is also what the counter
         style on .colist uses, so marker and list entry match.

         The comment marker in front goes too -- asciidoctor drops `//`, `#`,
         `--`, `;;` and an XML `` wrapper when it draws callouts as
         icons; the comment hides the marker from a compiler, it is not for the
         reader. Only the marker and one space, so a padded column survives.

         XML FIRST, because `<!--` ends in `--` and the line-comment rule
         below would otherwise eat half of it and leave `<!` behind. */ -}}
  {{- $html = replaceRE `<!--(\([0-9]+\))-->` "$1" $html -}}
  {{- $html = replaceRE `(?://|#|--|;;) ?()` "$1" $html -}}

  {{- /* ①-⑳, indexed by the marker's own number. Past 20 the parenthesised
         form stays -- Unicode's circled digits stop being one run of the same
         design shortly after (⑴ and ⒈ are separate series). */ -}}
  {{- $circled := split "①②③④⑤⑥⑦⑧⑨⑩⑪⑫⑬⑭⑮⑯⑰⑱⑲⑳" "" -}}
  {{- range $marker := findRE `\([0-9]+\)` $html -}}
    {{- $n := int (replaceRE `[^0-9]` "" $marker) -}}
    {{- if and (ge $n 1) (le $n (len $circled)) -}}
      {{- $html = replace $html $marker (index $circled (sub $n 1)) -}}
    {{- else -}}
      {{- $html = replace $html $marker (printf "(%d)" $n) -}}
    {{- end -}}
  {{- end -}}

{{- end -}}

{{- return ($html | safeHTML) -}}
