Posts can have multiple authors (`authors:` list); falls back to the legacy single `author:` slug so not-yet-reconverted posts still show a byline. */}} {{ $authors := .Params.authors }} {{ if and (not $authors) .Params.author }}{{ $authors = slice .Params.author }}{{ end }} {{ range $i, $slug := $authors }} {{- if $i }}, {{ end -}} {{- $author := partial "author-page.html" $slug -}} {{- if $author -}} {{ $author.Title }} {{- else -}} {{ $slug }} {{- end -}} {{ end }}
