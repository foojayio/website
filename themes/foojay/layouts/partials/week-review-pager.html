{{- /* The week pager, same look as pagination.html but every button links to a week's own dated URL, so a link keeps pointing at that week while the latest one moves on. Newest on the left, like page 1. Takes {page: the week page, position: "top"|"bottom"}. */ -}} {{- $weeks := (.page.Parent.RegularPages.ByParam "weekStart").Reverse -}} {{- $cur := 0 -}} {{- range $i, $w := $weeks }}{{ if eq $w.RelPermalink $.page.RelPermalink }}{{ $cur = $i }}{{ end }}{{ end -}} {{- $last := sub (len $weeks) 1 -}} {{- $size := 5 -}} {{- $from := sub $cur 2 -}} {{- if lt $from 0 }}{{ $from = 0 }}{{ end -}} {{- $to := add $from (sub $size 1) -}} {{- if gt $to $last }}{{ $to = $last }}{{ $from = sub $to (sub $size 1) }}{{ end -}} {{- if lt $from 0 }}{{ $from = 0 }}{{ end -}}
{{- if gt $cur 0 }} « {{- else }} « {{- end }}
    {{- range $i := seq $from $to }} {{- $w := index $weeks $i }} {{- $label := (time.AsTime (printf "%sT00:00:00Z" $w.Params.weekStart)).Format "2 Jan" }} {{- $d := sub $i $cur }}{{ if lt $d 0 }}{{ $d = sub 0 $d }}{{ end }} {{- if eq $i $cur }}
  • {{ $label }}
    • {{- else }} {{ $label }} {{- end }} {{- end }}
{{- if lt $cur $last }} » {{- else }} » {{- end }}
