{{- /* How long a list of Quick Start pages takes, in whole minutes. Takes the pages; returns 0 when none of them says. The hub page's "under N minutes" promise is DERIVED from the steps' own mm:ss `duration:` rather than typed next to them, so re-timing a step or adding a twelfth moves it. `math.Floor`, not round: rounding 29:52 up to 30 would make a true claim read as a false one. */ -}} {{- $seconds := 0 -}} {{- range . -}} {{- with .Params.duration -}} {{- $parts := split . ":" -}} {{- if eq (len $parts) 2 -}} {{- /* LEADING ZEROS OFF BEFORE `int`. Hugo's `int` is strconv.ParseInt with base 0, which reads "08" as octal and FAILS THE WHOLE BUILD ("unable to cast \"08\"") -- and mm:ss is written with them. The `default "0"` is for "00", which trims away to nothing. */ -}} {{- $m := int (strings.TrimLeft "0" (index $parts 0) | default "0") -}} {{- $s := int (strings.TrimLeft "0" (index $parts 1) | default "0") -}} {{- $seconds = add $seconds (add (mul 60 $m) $s) -}} {{- end -}} {{- end -}} {{- end -}} {{- return (int (math.Floor (div $seconds 60.0))) -}}
