{{- /* Link render hook. Three jobs: 1. Absolute foojay.io links in migrated content (e.g. "See also" links) point at the live WordPress site -- rewrite them to this site's path. 2. Root-relative internal links get the baseURL subpath (relURL only prefixes it when the path doesn't already start with "/"). 3. Links to any other host are external -> open in a new tab. String checks (not urls.Parse, which errors on malformed migrated URLs). */ -}} {{- $orig := .Destination -}} {{- $dest := $orig -}} {{- $isAbs := or (hasPrefix $orig "http://") (hasPrefix $orig "https://") (hasPrefix $orig "//") -}} {{- /* Host-anchored so foojay.social and query strings don't false-match. */ -}} {{- $internal := or (hasPrefix $orig "https://foojay.io") (hasPrefix $orig "http://foojay.io") (hasPrefix $orig "https://www.foojay.io") (hasPrefix $orig "http://www.foojay.io") (hasPrefix $orig "//foojay.io") -}} {{- $external := and $isAbs (not $internal) -}} {{- if $internal -}} {{- $dest = replaceRE "^(https?:)?//(www\\.)?foojay\\.io" "" $dest -}} {{- end -}} {{- if not (or $external (hasPrefix $dest "http") (hasPrefix $dest "//") (hasPrefix $dest "mailto:") (hasPrefix $dest "#") (hasPrefix $dest "data:")) -}} {{- $dest = $dest | strings.TrimPrefix "/" | relURL -}} {{- end -}} {{ .Text | safeHTML }}
