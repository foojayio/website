{{- /* Link render hook. Two jobs: 1. Root-relative internal links get the baseURL subpath (relURL only prefixes it when the path doesn't already start with "/"). 2. Links to any host other than foojay.io are treated as external and opened in a new tab (target="_blank" + safe rel). Relative links, anchors, mailto: and foojay.io / *.foojay.io stay in the same tab. */ -}} {{- $dest := .Destination -}} {{- $isAbs := or (hasPrefix $dest "http://") (hasPrefix $dest "https://") (hasPrefix $dest "//") -}} {{- /* External = absolute and not a foojay.io URL. String check (not urls.Parse, which errors on malformed destinations found in migrated content). */ -}} {{- $external := and $isAbs (not (in $dest "foojay.io")) -}} {{- if not (or $isAbs (hasPrefix $dest "mailto:") (hasPrefix $dest "#") (hasPrefix $dest "data:")) -}} {{- $dest = $dest | strings.TrimPrefix "/" | relURL -}} {{- end -}} {{ .Text | safeHTML }}
