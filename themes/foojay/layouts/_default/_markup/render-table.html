{{- /* A TABLE WHOSE HEADER ROW IS ENTIRELY EMPTY IS RENDERED WITHOUT ONE. GFM cannot write a headerless table -- the delimiter row is what makes a table a table, and it only comes after a header -- but many imported tables are headerless (legends, note boxes, annotations). Those are stored with an EMPTY header row, the only valid GFM shape that keeps every data row a data row, and this hook takes it back out. Without it the reader gets a blank strip: `.prose th` carries a bottom border and heading weight, so an empty header is visible furniture. Derived from the content -- a real header is kept, a blank one is dropped, and no frontmatter chooses. The rest of the markup is what Goldmark's own renderer emits, so `.prose table` is unaffected. */ -}} {{- $hasHeader := false -}} {{- range .THead -}} {{- range . -}} {{- if gt (len (strings.TrimSpace (.Text | plainify))) 0 -}}{{- $hasHeader = true -}}{{- end -}} {{- end -}} {{- end -}} {{- if $hasHeader }} {{- range .THead }} {{- range . }} {{ .Text }} {{- end }} {{- end }} {{- end }} {{- range .TBody }} {{- range . }} {{ .Text }} {{- end }} {{- end }}
