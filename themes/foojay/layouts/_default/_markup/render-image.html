{{- /* Resolve the image src: 1. Page-bundle resource (a plain filename like foo.png co-located with index.md) -> its .RelPermalink (already baseURL-subpath correct, and it publishes the image to the post's URL). This is the author-friendly co-located layout. 2. Root-relative static path (/images/... from pages/pedia) -> relURL, stripping the leading slash so the subpath is applied. 3. External / data URLs -> untouched. */ -}} {{- /* htmlUnescape for the same reason as render-link.html: overriding the renderer means reproducing the entity decoding goldmark would have done. */ -}} {{- $dest := .Destination | htmlUnescape -}} {{- $abs := or (hasPrefix $dest "http://") (hasPrefix $dest "https://") (hasPrefix $dest "//") (hasPrefix $dest "data:") -}} {{- /* width/height whenever the image is a page-bundle resource Hugo can measure. Not cosmetic: without them the browser reserves no space, so every image shoves the text below it down as it decodes. The CSS caps them (`.prose img { max-width: 100% }`) and the browser scales the reserved box by the same ratio, so this fixes the shift without fixing the size. ONLY A RASTER CAN BE MEASURED. Hugo reports an SVG as ResourceType "image" too, but .Width on one halts the build -- so the MEDIA SUBTYPE is what is tested, not the resource type. */ -}} {{- $w := 0 -}} {{- $h := 0 -}} {{- if not $abs -}} {{- with .Page.Resources.GetMatch $dest -}} {{- $dest = .RelPermalink -}} {{- if and (eq .ResourceType "image") (in (slice "png" "jpeg" "gif" "webp" "bmp" "tiff") .MediaType.SubType) -}} {{- $w = .Width -}} {{- $h = .Height -}} {{- end -}} {{- else -}} {{- $dest = $dest | strings.TrimPrefix "/" | relURL -}} {{- end -}} {{- end -}} {{- /* NO TRAILING NEWLINE, same reason as render-link.html: an image is inline, so a newline after the tag collapses to a space between it and whatever follows. */ -}}
