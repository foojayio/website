{{- /* Resolve the image src: 1. Page-bundle resource (a plain filename like foo.png co-located with index.md) -> its .RelPermalink (already baseURL-subpath correct, and it publishes the image to the post's URL). This is the author-friendly co-located layout. 2. Root-relative static path (/images/... from pages/pedia) -> relURL, stripping the leading slash so the subpath is applied. 3. External / data URLs -> untouched. */ -}} {{- /* htmlUnescape for the same reason as render-link.html: overriding the renderer means reproducing the entity decoding goldmark would have done. */ -}} {{- $dest := .Destination | htmlUnescape -}} {{- $abs := or (hasPrefix $dest "http://") (hasPrefix $dest "https://") (hasPrefix $dest "//") (hasPrefix $dest "data:") -}} {{- if not $abs -}} {{- with .Page.Resources.GetMatch $dest -}} {{- $dest = .RelPermalink -}} {{- else -}} {{- $dest = $dest | strings.TrimPrefix "/" | relURL -}} {{- end -}} {{- end -}}
